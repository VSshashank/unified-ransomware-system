# Dependencies of the measurement scripts in this directory.
#
# These are not service dependencies. No container installs them and nothing in
# services/ imports them; they exist so that an independent reader can rebuild
# the corpus and re-run the measurements the thesis cites.
#
# This file was added in Phase 8 because scripts/verify_reproduction.py ran the
# reproducibility appendix from a clean checkout and the corpus rebuild failed
# with ModuleNotFoundError: No module named 'PIL'. Pillow was installed on the
# author's machine and declared nowhere, so every benign-cost figure in the
# thesis rested on a dependency a reader had no way to discover.

Pillow==12.3.0      # scripts/build_benign_corpus.py - the photographic and
                    # noise-image corpus cells. The corpus manifest's per-file
                    # hashes are hashes of Pillow's output, so the version is
                    # pinned: a different encoder writes different bytes and
                    # --verify would correctly reject the rebuild.

psutil==7.2.2       # scripts/load_test.py - process and memory sampling.
                    # Pinned to the same version services/response declares, so
                    # a single environment can run both.
